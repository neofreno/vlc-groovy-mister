cmake_minimum_required(VERSION 3.22)
project(groovy_transport_tests LANGUAGES C CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 14)
add_library(api_under_test STATIC ../api/groovymister.cpp ../api/lz4/lz4.c ../api/lz4/lz4hc.c)
if(WIN32)
    target_compile_definitions(api_under_test PUBLIC NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(api_under_test PUBLIC ws2_32)
endif()
add_executable(protocol_tests protocol_tests.cpp)
add_executable(transport_safety_tests transport_safety_tests.cpp)
add_executable(api_client api_client.cpp)
add_executable(api_lifecycle api_lifecycle.cpp)
add_executable(resource_lifecycle_tests resource_lifecycle_tests.cpp)
foreach(profile standard xdp wifi)
    add_executable(profile_${profile}_tests receiver_profile_tests.cpp)
endforeach()
target_compile_definitions(profile_xdp_tests PRIVATE _AF_XDP)
target_compile_definitions(profile_wifi_tests PRIVATE _WIFI_MODE)
target_link_libraries(api_client PRIVATE api_under_test)
target_link_libraries(api_lifecycle PRIVATE api_under_test)
foreach(test protocol_tests transport_safety_tests api_lifecycle resource_lifecycle_tests profile_standard_tests profile_xdp_tests profile_wifi_tests)
    # Tests use assert, including in release builds.
    if(MSVC)
        target_compile_options(${test} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${test} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${test} COMMAND ${test})
endforeach()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME api_udp_loopback COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_api_transport.py $<TARGET_FILE:api_client>)
set_tests_properties(api_udp_loopback PROPERTIES TIMEOUT 240)
add_test(NAME core_cleanup_hooks COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test_core_cleanup_hooks.py
    ${CMAKE_CURRENT_SOURCE_DIR}/../hps_linux/src/fpga_io.cpp)
