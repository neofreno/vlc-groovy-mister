cmake_minimum_required(VERSION 3.22)
project(reconnect_loopback LANGUAGES C CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(GROOVY_API_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../Groovy_MiSTer/api" CACHE PATH "Groovy API sources")
add_library(api STATIC "${GROOVY_API_DIR}/groovymister.cpp" "${GROOVY_API_DIR}/lz4/lz4.c" "${GROOVY_API_DIR}/lz4/lz4hc.c")
target_include_directories(api PUBLIC "${GROOVY_API_DIR}")
if(WIN32)
    target_compile_definitions(api PUBLIC NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(api PUBLIC ws2_32)
endif()
add_executable(reconnect_client reconnect_client.cpp)
target_link_libraries(reconnect_client PRIVATE api)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME reconnect_udp_loopback COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/test_reconnect.py" $<TARGET_FILE:reconnect_client>)
set_tests_properties(reconnect_udp_loopback PROPERTIES TIMEOUT 90)
add_executable(audio_session_client audio_session_client.cpp)
target_link_libraries(audio_session_client PRIVATE api)
add_test(NAME audio_session_udp_loopback COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/test_audio_session.py" $<TARGET_FILE:audio_session_client>)
set_tests_properties(audio_session_udp_loopback PROPERTIES TIMEOUT 90)
add_executable(joypad_session_client joypad_session_client.cpp ../../src/groovymister_wrapper.cpp)
target_compile_definitions(joypad_session_client PRIVATE GROOVYMISTER_WIN32_STATIC)
target_link_libraries(joypad_session_client PRIVATE api)
add_test(NAME joypad_session_udp_loopback COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/test_joypad_session.py" $<TARGET_FILE:joypad_session_client>)
set_tests_properties(joypad_session_udp_loopback PROPERTIES TIMEOUT 20)
