cmake_minimum_required(VERSION 3.22)
project(video_stream_tests LANGUAGES CXX)
option(VIDEO_TEST_ASAN "Check memory accesses with AddressSanitizer" OFF)
enable_testing()
add_executable(video_stream_tests video_stream_tests.cpp)
add_executable(audio_timeline_tests audio_timeline_tests.cpp)
add_executable(stream_startup_tests stream_startup_tests.cpp)
add_executable(stream_health_tests stream_health_tests.cpp)
add_executable(stream_audio_format_tests stream_audio_format_tests.cpp)
add_executable(video_ack_metrics_tests video_ack_metrics_tests.cpp)
add_executable(audio_ring_submit_tests audio_ring_submit_tests.cpp)
add_executable(video_conversion_compare video_conversion_compare.cpp)
add_executable(joypad_controls_tests joypad_controls_tests.cpp)
add_executable(joypad_vlc_tests joypad_vlc_tests.cpp ../src/joypad.cpp)
target_include_directories(joypad_vlc_tests PRIVATE joypad_stubs)
target_compile_definitions(joypad_vlc_tests PRIVATE NOMINMAX GROOVYMISTER_WIN32_STATIC)
add_executable(interface_timer_tests interface_timer_tests.cpp)
target_include_directories(interface_timer_tests PRIVATE joypad_stubs)
foreach(test video_stream_tests audio_timeline_tests stream_startup_tests stream_health_tests stream_audio_format_tests video_ack_metrics_tests audio_ring_submit_tests video_conversion_compare joypad_controls_tests joypad_vlc_tests interface_timer_tests)
target_compile_features(${test} PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(${test} PRIVATE /W4 /WX /EHsc)
    if(VIDEO_TEST_ASAN)
        target_compile_options(${test} PRIVATE /fsanitize=address /Zi)
        target_link_options(${test} PRIVATE /INCREMENTAL:NO)
    endif()
endif()
add_test(NAME ${test} COMMAND ${test})
if(MSVC AND VIDEO_TEST_ASAN)
    get_filename_component(compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set_tests_properties(${test} PROPERTIES
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${compiler_bin}")
endif()
endforeach()
set_tests_properties(stream_startup_tests PROPERTIES TIMEOUT 10)
set_tests_properties(joypad_controls_tests PROPERTIES TIMEOUT 10)
set_tests_properties(joypad_vlc_tests PROPERTIES TIMEOUT 15)
set_tests_properties(interface_timer_tests PROPERTIES TIMEOUT 10)
